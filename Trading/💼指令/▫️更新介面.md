  - 頁面底部必須保留 TAB 切換的 JavaScript 邏輯：
    ```javascript
    document.querySelectorAll('.stock-tab').forEach(function(tab) {
      tab.addEventListener('click', function() {
        document.querySelectorAll('.stock-tab').forEach(function(t) { t.classList.remove('active'); });
        this.classList.add('active');
        var targetId = this.getAttribute('data-target');
        document.querySelectorAll('.stock-section').forEach(function(sec) { sec.style.display = 'none'; });
        var targetSec = document.getElementById(targetId);
        if (targetSec) { targetSec.style.display = 'block'; }
      });
    });
    ```
